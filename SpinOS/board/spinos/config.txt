# SpinOS: Raspberry Pi 3B. The projector must stay black from power-on.
start_file=start.elf
fixup_file=fixup.dat
kernel=Image
arm_64bit=1

# no rainbow square, no boot delay
disable_splash=1
boot_delay=0

# HDMI: KMS driver (SpinProj draws through DRM), output on even if the projector is switched on later
dtoverlay=vc4-kms-v3d
hdmi_force_hotplug=1
disable_overscan=1
gpu_mem=64

# GPIO14/15 (pins 8, 10) are the three-wire link to the ESP32 (or the USB cable, either works):
# the PL011 UART goes there, Bluetooth is not used, and no console talks on it
enable_uart=1
dtoverlay=disable-bt

# not needed
dtparam=audio=off
